เอาโมเดลครั้งที่ 4 มาแก้ตามหลักฐานที่ validation ครั้งที่ 5 เจอ แล้วห่อเป็นฟังก์ชัน run_one(seed) เพื่อรันซ้ำได้

# ===== โมเดล dock เวอร์ชันแก้ตามผล Validation ครั้งที่ 5 =====
import simpy, random, statistics as st, math
from scipy import stats

SHIFT, NOON    = 480, 240        # กะ 8 ชม. · เส้นแบ่งเช้า/บ่าย  (validation ครั้งที่ 5: ผ่าน)
IAT_AM, IAT_PM = 10, 6           # รถเข้าเฉลี่ยทุก 10 / 6 นาที    (validation ครั้งที่ 5: ผ่าน)

# [แก้ที่ 1] รถ 2 ประเภท — histogram จากการจับเวลา 40 คัน แยกเป็น 2 ก้อนชัดเจน
SVC_NORMAL  = (12, 15, 23)       # รถทั่วไป 30 คัน  · Triangular(ต่ำสุด, ฐานนิยม, สูงสุด)
SVC_TRAILER = (34, 40, 51)       # รถพ่วง   10 คัน  · โมเดลเดิม Tri(12,20,34) สร้างค่านี้ไม่ได้เลย
P_TRAILER   = 0.25               # 10 จาก 40 คัน

FREE_MIN, FINE_PER_HR = 45, 2500 # เกิน 45 นาที ปรับ 2,500 บาท/ชม. ตามส่วน
OT_PER_HR             = 600      # ค่าแรงทีมขนถ่ายนอกเวลา (สมมติ — ต้องยืนยันกับ HR)

def make_day(seed):
    """สร้าง 'วัน' 1 วันจาก seed: รถแต่ละคันมาเมื่อไร ประเภทอะไร ใช้เวลาขนถ่ายเท่าไร
       [สำคัญ] แยกการสุ่มออกมาก่อน เพื่อให้ทุกทางเลือกเจอวันเดียวกันเป๊ะ — นี่คือ CRN ที่แท้จริง"""
    rng, t, trucks = random.Random(seed), 0.0, []
    while True:
        t += rng.expovariate(1/(IAT_AM if t < NOON else IAT_PM))
        if t >= SHIFT: break                      # ปิดรับรถเมื่อหมดกะ
        lo, mode, hi = SVC_TRAILER if rng.random() < P_TRAILER else SVC_NORMAL
        trucks.append((t, rng.triangular(lo, hi, mode)))   # Python รับ (ต่ำสุด, สูงสุด, ฐานนิยม)
    return trucks

def run_one(seed, base_docks=3, extra_from=None):
    """รัน 1 replication = 1 กะ · คืน KPI ของกะนั้น"""
    trucks = make_day(seed)
    env  = simpy.Environment()
    dock = simpy.Resource(env, capacity=base_docks + (1 if extra_from is not None else 0))
    out  = []

    def hold_extra(env):                      # ช่องที่ 4 ยังไม่เปิด → ยึดไว้ก่อนจนถึงนาทีที่กำหนด
        with dock.request() as req:
            yield req
            yield env.timeout(extra_from)

    def truck(env, arrive, svc):
        yield env.timeout(arrive)             # รอจนถึงเวลาที่รถคันนี้มาถึง
        with dock.request() as req:
            yield req
            yield env.timeout(svc)
        out.append((arrive, env.now))

    if extra_from is not None: env.process(hold_extra(env))
    for arrive, svc in trucks: env.process(truck(env, arrive, svc))

    # [แก้ที่ 2] ไม่ใส่ until=SHIFT — ต้องปล่อยให้รถที่รอคิวอยู่ขนถ่ายจนเสร็จ (คือ OT ของจริง)
    #           โมเดลครั้งที่ 4 ตัดรถกลุ่มนี้ทิ้ง ซึ่งเป็นกลุ่มที่รอนานที่สุด → ตัวเลขสวยกว่าความจริง
    env.run()

    ta   = [end - arr for arr, end in out]
    fine = sum(max(0, x - FREE_MIN)/60*FINE_PER_HR for x in ta)       # ค่าปรับ detention
    ot_h = max(0, max(end for arr, end in out) - SHIFT)/60            # ชั่วโมงที่เลยกะ
    return {"n": len(out), "ta": st.mean(ta),
            "over45": 100*sum(x > FREE_MIN for x in ta)/len(ta),
            "late"  : sum(1 for arr, end in out if end > SHIFT),
            "fine"  : fine, "ot_h": ot_h, "ot_cost": ot_h*OT_PER_HR}

3 Terminating vs Non-terminating — เลือกผิด วิเคราะห์ผิดทั้งหมด

คำถามแรกก่อนวิเคราะห์ผลเสมอ: ระบบนี้มี "จุดเริ่มและจุดจบตามธรรมชาติ" ไหม

⏹️ Terminating (มีจุดจบ)

  • มีเงื่อนไขเริ่ม–จบชัดเจน · เริ่มจากระบบว่าง
  • ตัวอย่าง: กะทำงานของ dock (8 ชม.) · คลังเปิด–ปิด · การหยิบของ 1 วัน
  • 1 replication = 1 กะ · แต่ละกะเป็นอิสระต่อกัน
  • ใช้สถิติมาตรฐานได้เลย (x̄ ± t·s/√n)
  • ไม่ต้องตัด warm-up เพราะการเริ่มจากว่างคือความจริง

♾️ Non-terminating (ต่อเนื่อง)

  • เดินต่อเนื่อง สนใจพฤติกรรมระยะยาว (steady state)
  • ตัวอย่าง: สายพานลำเลียงที่เดิน 24 ชม. · ศูนย์คัดแยกพัสดุที่ไม่เคยหยุด
  • ช่วงแรกระบบยัง "ว่างผิดปกติ" → ต้องตัด warm-up ทิ้ง
  • รันยาว 1 ครั้งแล้วแบ่งเป็นชุด (batch means) หรือรันหลายรอบยาว ๆ

🌡️ แล้ว "warm-up" ที่พูดถึงคืออะไร

Warm-up period (ช่วงอุ่นเครื่อง / transient period) คือ ช่วงตั้งแต่เริ่มรัน จนกว่าระบบจะเข้าสู่สภาวะคงตัว — ข้อมูลช่วงนี้สะท้อน สภาพเริ่มต้นที่เราตั้งเอง ไม่ใช่พฤติกรรมจริงของระบบ

เทียบให้เห็นภาพ อยากรู้ว่าแอร์ทำความเย็นได้กี่องศา แล้วเปิดแอร์ตอนห้องร้อน 35° · ถ้าจดอุณหภูมิตั้งแต่วินาทีแรก ค่าเฉลี่ยจะเพี้ยน เพราะ 20 นาทีแรกคือ "แอร์กำลังไล่ความร้อน" — 20 นาทีนั้นคือ warm-up

เกิดขึ้นเพราะโมเดลต้องเริ่มจากสภาพอะไรสักอย่าง และค่าที่ตั้งง่ายที่สุดคือ ว่างเปล่าไม่มีงานค้าง (empty and idle) ซึ่งระบบที่เดินต่อเนื่องไม่เคยเป็นแบบนั้นจริง → ช่วงแรกคิวสั้นผิดปกติ ค่าเฉลี่ยจึงต่ำกว่าความจริง เรียกว่า initialization bias
วิธีแก้คือ รันช่วงนั้นตามปกติ แต่ไม่นับข้อมูลของมัน (warm-up deletion) — เริ่มเก็บสถิติหลังจากกราฟนิ่งแล้ว · ไม่ใช่ "ข้ามไม่รัน" เพราะต้องปล่อยให้งานค้างสะสมจนสมจริงเสียก่อน

เคสของเราทั้งสองอันเป็น Terminating โมเดล dock (1 กะ) และโมเดลหยิบสินค้า (1 วัน) มีจุดเริ่ม–จบชัดเจน · จึงไม่ต้องตัด warm-up และ 1 replication = 1 กะ/1 วัน · ทำให้การวิเคราะห์ง่ายกว่ามาก

# ===== เก็บผลหลาย replication แล้วสร้างช่วงความเชื่อมั่น =====
def ci(values, conf=0.95):
    """คืน (ค่าเฉลี่ย, SD, ครึ่งช่วง) — ครึ่งช่วง = t * s / sqrt(n)"""
    m, s, n = st.mean(values), st.stdev(values), len(values)
    t = stats.t.ppf(0.5 + conf/2, n - 1)          # ค่าวิกฤต t ที่ df = n-1
    return m, s, t*s/math.sqrt(n)

def summary(base_docks=3, extra_from=None, n_rep=100, seed0=42):
    """รัน n_rep กะ ด้วย seed ต่อเนื่อง แล้วสรุปทุก KPI พร้อมช่วงความเชื่อมั่น"""
    runs = [run_one(seed0 + i, base_docks, extra_from) for i in range(n_rep)]
    out  = {}
    for key in ["ta", "over45", "late", "fine", "ot_h"]:
        m, s, hw = ci([r[key] for r in runs])
        out[key] = {"mean": m, "sd": s, "hw": hw, "rel": hw/m*100 if m else 0}
    return runs, out

🎲 Common Random Numbers (CRN) — เทคนิคที่ทำให้เทียบได้ยุติธรรม

❌ ไม่ใช้ CRN

  • ทางเลือก A เจอรถ 62 คัน · B เจอรถ 71 คัน
  • ความต่างที่เห็น = ผลของทางเลือก + ความต่างของวัน
  • ต้องรันเยอะมากกว่าจะเห็นความต่างชัด

✅ ใช้ CRN

  • ทั้ง A และ B เจอรถชุดเดียวกันเป๊ะทุกกะ
  • ความต่างที่เห็น = ผลของทางเลือกล้วน ๆ
  • ความแปรปรวนของผลต่างลดลงมาก → สรุปได้เร็วกว่า
วิธีทำที่ถูกต้องในโค้ดของเรา ไม่ใช่แค่ตั้ง random.seed(i) แล้วรันแต่ละทางเลือก — วิธีนั้น seed จะ "หลุดคู่" ทันทีที่สองทางเลือกเรียกตัวสุ่มไม่เท่ากัน (เช่น มีช่องมากกว่า → ลำดับเหตุการณ์ต่างกัน) รถคันที่ 30 ของ A กับของ B จะกลายเป็นคนละคัน
✅ วิธีที่โค้ดในสไลด์ใช้: สุ่ม "วัน" ให้เสร็จก่อนด้วย random.Random(seed) ใน make_day(seed) แล้วค่อยส่งรถชุดเดียวกันนั้นเข้าไปให้ทุกทางเลือกจำลอง — แยกการสุ่มออกจากการจำลอง จึงการันตีได้ว่าเจอ "วันเดียวกันเป๊ะ" เสมอ

📊 Paired-t — ดูที่ "ผลต่าง" ไม่ใช่ที่ค่าแต่ละตัว

  1. รันทั้งสองทางเลือกด้วย seed เดียวกัน กะที่ 1, 2, ..., n
  2. คำนวณ ผลต่างรายกะ: Di = KPIB,i − KPIA,i
  3. สร้างช่วงความเชื่อมั่นของ D̄ ด้วยสูตรเดิม: D̄ ± t·sD/√n
  4. อ่านผล: ถ้าช่วงไม่คร่อม 0 → ต่างกันอย่างมีนัยสำคัญ · ถ้าคร่อม 0 → ยังสรุปไม่ได้
"ยังสรุปไม่ได้" ≠ "เท่ากัน" ถ้าช่วงคร่อม 0 มี 2 ความเป็นไปได้: ① ทั้งสองทางเลือกใกล้เคียงกันจริง ② รันไม่พอ · ทางออก: รันเพิ่มแล้วดูว่าช่วงแคบลงจนพ้น 0 ไหม · และถ้าต่างกันน้อยมากจริง ๆ นั่นก็เป็นคำตอบทางธุรกิจที่มีค่า — เลือกทางที่ถูกกว่าได้เลย

⚖️ ทดลอง: CRN ช่วยได้จริงแค่ไหน — เปิดกับปิด ในคลิกเดียว

ทำไมต้องมีวิดเจ็ตนี้ทั้งที่มีโค้ดอยู่แล้ว: โค้ดในสไลด์ถัดไปใช้ CRN อย่างเดียว จึงไม่มีทางแสดงให้เห็นว่า "ถ้าไม่ใช้จะเป็นยังไง" · ปุ่มนี้รันทั้งสองแบบด้วย n เท่ากัน แล้ววางเทียบความกว้างของช่วงให้ดู

# ===== เทียบ 3 ทางเลือกด้วย CRN + paired-t =====
N_REP    = 100
SCENARIO = {"A": dict(base_docks=3),                    # ไม่ทำอะไร
            "B": dict(base_docks=4),                    # ทีมเสริมเปิดช่องที่ 4 ตลอดกะ
            "C": dict(base_docks=3, extra_from=240)}    # ทีมเสริมเฉพาะช่วงบ่าย

# CRN = ใช้ seed ชุดเดียวกัน (42..141) กับทุกทางเลือก -> ทุกทางเลือกเจอ "วัน" เดียวกัน
RES = {k: summary(n_rep=N_REP, seed0=42, **kw)[0] for k, kw in SCENARIO.items()}

def paired(key, x, y):
    """ผลต่างรายกะ D_i = KPI(x,i) - KPI(y,i) แล้วสร้าง CI ของ D-bar"""
    d = [RES[x][i][key] - RES[y][i][key] for i in range(N_REP)]
    m, s, hw = ci(d)
    return m, m - hw, m + hw

for x, y in [("B", "A"), ("C", "A"), ("B", "C")]:
    m, lo, hi = paired("ta", x, y)
    verdict = "ต่างกันอย่างมีนัยสำคัญ" if lo*hi > 0 else "ยังสรุปไม่ได้ (ช่วงคร่อม 0)"
    print(f"{x} - {y}: turnaround {m:+6.2f} นาที  ช่วง 95% [{lo:+6.2f}, {hi:+6.2f}]  -> {verdict}")

B - C: turnaround  -3.09 นาที  ช่วง 95% [ -3.96,  -2.22]  -> ต่างกันอย่างมีนัยสำคัญ
อ่านผล ทั้ง B และ C ดีกว่า "ไม่ทำอะไร" อย่างชัดเจน (ช่วงไม่คร่อม 0 และห่างจาก 0 มาก) · ส่วน B − C = −3.09 นาที ช่วง [−3.96, −2.22] ก็ไม่คร่อม 0 เช่นกัน → B เร็วกว่า C จริง ไม่ใช่บังเอิญ
